```typescript
export type WriteGateStatus = "pending" | "confirmed" | "rejected";

// Task types map to collection names. Extend this as new skills are added.
export const TASK_COLLECTIONS: Record<string, string> = {
  "email-triage": "task-email-triage",
  "proposal-gen": "task-proposal-gen",
  "form-gen": "task-form-gen",
  "lead-enrich": "task-lead-enrich",
  "check-in": "task-check-in",
};

export type TaskType = keyof typeof TASK_COLLECTIONS;

export interface OutcomeRecord {
  task_type: string;
  input_summary: string;       // What was the task about (max ~200 chars)
  output_summary: string;      // What was produced (max ~500 chars)
  success: boolean;
  error_type?: string;         // "timeout" | "api_failure" | "wrong_output" | "hallucination"
  error_message?: string;
  token_usage: { input: number; output: number; cost_usd: number };
  model_tier: string;          // "ollama" | "haiku" | "sonnet" | "opus"
  operator_correction?: string;
  patterns_observed?: string;
  contact_id?: string;         // If task was client-related
  session_id: string;
  timestamp: string;           // ISO 8601
}

export interface OutcomePayload extends OutcomeRecord {
  status: WriteGateStatus;
  indexed_at: string;          // ISO 8601
  narrative: string;           // Embedding source text
  frequency?: number;          // Dedup counter for routine successes
}

export interface TaskRagConfig {
  task_types: Record<string, { collection: string; recall_limit: number; recall_threshold: number }>;
  write_gates: {
    promotion_ttl_days: number;
    promotion_schedule: string;  // cron expression
    require_success_for_auto_promote: boolean;
  };
  dedup: {
    similarity_threshold: number; // 0.95 default
    enabled: boolean;
  };
}
```

**1. Create `~/.openclaw/src/task-rag/outcome-indexer.ts`:**

Imports:
- `embed`, `createQdrantClient`, `createOllamaClient` from `../memory/vector-search.js`
- `TASK_COLLECTIONS`, `OutcomeRecord`, `OutcomePayload` from `./types.js`

**Helper: `deterministicOutcomeId(taskType: string, sessionId: string, timestamp: string): number`**
- Include `taskType` in the hash input to avoid cross-collection ID collisions (Pitfall 6 from research)
- Hash formula: same djb2 hash as vector-search.ts `deterministicId()` but with input = `${taskType}:${sessionId}:${timestamp}`
- Return `Math.abs(hash) % 2147483647 || 1`

**Function: `buildOutcomeNarrative(outcome: OutcomeRecord): string`**
- Build a concise structured narrative for embedding (NOT the full LLM output):
  - `Task: ${outcome.task_type}`
  - `Input: ${outcome.input_summary}`
  - `Result: ${outcome.success ? "SUCCESS" : "FAILURE"}`
  - If failure: `Error: ${outcome.error_type} - ${outcome.error_message}`
  - `Output: ${outcome.output_summary}`
  - If correction: `Correction: ${outcome.operator_correction}`
  - If pattern: `Pattern: ${outcome.patterns_observed}`
- Join with `. ` separator
- Export this function (used by tests and Plan 06-02)

**Function: `shouldIndex(outcome: OutcomeRecord): boolean`**
- Return `true` if any of:
  - `success === false` (errors are always valuable)
  - `operator_correction` is truthy (corrections are always valuable)
  - `patterns_observed` is truthy (novel patterns are valuable)
- Return `false` for routine successes with no correction and no observed pattern
- This prevents outcome index pollution (Pitfall 1 from research)

**Function: `checkDuplicate(taskType: string, narrative: string, qdrant?: QdrantClient, ollama?: Ollama): Promise<{ isDuplicate: boolean; existingPointId?: number | string }>`**
- Load dedup config from `~/.openclaw/config/task-rag.yaml`
- If `dedup.enabled` is false, return `{ isDuplicate: false }`
- Embed the narrative
- Search the task collection with `limit: 1`, `score_threshold: dedup.similarity_threshold` (default 0.95)
- If a result with `success: true` exists above threshold, return `{ isDuplicate: true, existingPointId: result.id }`
- Otherwise return `{ isDuplicate: false }`

**Function: `indexOutcome(outcome: OutcomeRecord, options?: { qdrant?: QdrantClient; ollama?: Ollama }): Promise<{ indexed: boolean; pointId?: number; reason?: string }>`**
- If `!shouldIndex(outcome)`, return `{ indexed: false, reason: "routine_success" }`
- Build narrative via `buildOutcomeNarrative()`
- Check duplicate via `checkDuplicate()`. If duplicate, increment `frequency` counter on existing point via `qdrant.setPayload()` and return `{ indexed: false, reason: "duplicate" }`
- Embed the narrative
- Generate deterministic point ID via `deterministicOutcomeId()`
- Upsert into the task-specific collection with payload: all OutcomeRecord fields + `status: "pending"`, `indexed_at: new Date().toISOString()`, `narrative`, `frequency: 1`
- Log to daily log: `{ type: "action", summary: "Task RAG: indexed ${outcome.task_type} outcome (${outcome.success ? 'success' : 'failure'})" }`
- Return `{ indexed: true, pointId }`

**Function: `postExecutionHook(taskType, input, result, usage, sessionId, operatorCorrection?): Promise<void>`**
- Convenience wrapper that constructs an OutcomeRecord from individual parameters and calls `indexOutcome()`
- Cap `result.output` at 500 chars for `output_summary`
- This is the function skills call after execution

**1. Create `~/.openclaw/src/task-rag/write-gates.ts`:**

Imports:
- `createQdrantClient` from `../memory/vector-search.js`
- `TASK_COLLECTIONS`, `OutcomePayload`, `WriteGateStatus` from `./types.js`
- `logEntry` from `../memory/daily-log.js`

- `readFileSync` from `fs`, `join` from `path`, `parse as parseYaml` from `yaml` -- for config loading

**Function: `promotePendingEntries(qdrant?: QdrantClient): Promise<number>`**
- Load write gate config from `~/.openclaw/config/task-rag.yaml`
- Calculate cutoff date: `new Date(Date.now() - config.write_gates.promotion_ttl_days * 24 * 60 * 60 * 1000).toISOString()`
- For each collection in TASK_COLLECTIONS:
  - `qdrant.scroll()` with filter: `status = "pending"` AND `indexed_at < cutoff`
  - For each point found:
    - Quality gate (Pitfall 3 from research): Only promote if `payload.success === true` OR `payload.operator_correction` is truthy. Skip failed outcomes without corrections -- they stay pending indefinitely.
    - If qualifies: `qdrant.setPayload(collectionName, { payload: { status: "confirmed" }, points: [point.id] })`
    - Increment promoted counter
- Log: `{ type: "action", summary: "Write gates: promoted ${promoted} pending outcomes to confirmed (${config.write_gates.promotion_ttl_days}-day TTL expired)" }`
- Return promoted count

**Function: `operatorValidateOutcome(pointId: number, taskType: string, decision: "confirm" | "reject", qdrant?: QdrantClient): Promise<void>`**
- Get collection name via `getCollectionName(taskType)` from collection-manager
- Map decision: `confirm` -> `"confirmed"`, `reject` -> `"rejected"`
- `qdrant.setPayload(collectionName, { payload: { status: newStatus, validated_by: "operator", validated_at: new Date().toISOString() }, points: [pointId] })`
- Log: `{ type: "interaction", summary: "Write gates: operator ${decision}ed outcome ${pointId} in ${taskType}" }`

**Function: `getPendingReviewSummary(qdrant?: QdrantClient): Promise<{ total: number; byTaskType: Record<string, number>; failedNeedingReview: number }>`**
- For each collection in TASK_COLLECTIONS:
  - `qdrant.scroll()` with filter: `status = "pending"`, limit 100
  - Count total pending per task type
  - Count entries where `success === false` and no `operator_correction` (these are the "stuck" ones that will never auto-promote)
- Return summary object (used by Phase 4 check-in engine for weekly review surfacing)

**Function: `scheduleWriteGatePromotion(): ReturnType<typeof schedule>`**
- Load config, get `write_gates.promotion_schedule` cron expression (default "0 2 * * *")
- `schedule(cronExpr, async () => { ... })` wrapping `promotePendingEntries()` with try/catch
- On error: log `{ level: "ERROR", component: "write-gates", action: "PROMOTION_FAILED" }` via stderr JSON
- Return the cron task (for testing and cleanup)
